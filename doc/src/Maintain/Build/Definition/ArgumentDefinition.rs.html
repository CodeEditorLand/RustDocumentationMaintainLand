93/// Represents parsed command-line arguments and environment variables that
94/// control the build.
95///
96/// This struct is generated by `clap` from the `Parser` derive macro and
97/// automatically parses command-line arguments and environment variables into
98/// a strongly-typed configuration object. Each field can be configured via:
99/// - Command-line arguments (long form, e.g., `--directory`)
100/// - Environment variables (as specified in the `env` attribute)
101/// - Default values (as specified in the `default_value` attribute)
102///
103/// Required fields must be provided, while optional fields can be omitted.
104/// The `Command` field is special in that it accepts all remaining arguments
105/// as the build command to execute.
106///
107/// # Field Descriptions
108///
109/// - **Directory**: The main directory of the project containing Cargo.toml
110/// - **Name**: The original base name of the project/package
111/// - **Prefix**: The prefix for the application's bundle identifier
112/// - **Browser**: Flag for browser-specific build aspects
113/// - **Bundle**: Flag for bundling-specific aspects
114/// - **Compile**: Flag for compile-specific aspects
115/// - **Clean**: Flag for cleaning-specific aspects
116/// - **Debug**: Flag for debug-specific aspects
117/// - **Dependency**: Information about a dependency (org/repo or boolean)
118/// - **Environment**: The Node.js environment (development, production, etc.)
119/// - **NodeVersion**: The Node.js sidecar version to bundle
120/// - **Command**: The build command and its arguments to execute
121#[derive(Parser, Debug, Clone)]
122#[clap(
123	author,
124	version,
125	about = "Prepares, builds, and restores project configurations."
126)]
127pub struct Argument {
128	/// The main directory of the project.
129	///
130	/// This field specifies the project directory that contains the
131	/// `Cargo.toml` and `tauri.conf.json` files. It can be set via:
132	/// - Command-line: `--directory <path>`
133	/// - Environment: `MOUNTAIN_DIR`
134	/// - Default: "Element/Mountain"
135	#[clap(long, env = DirEnv, default_value = DirectoryDefault)]
136	pub Directory: String,
137
138	/// The original base name of the project/package.
139	///
140	/// This field specifies the base name of the project, which serves as
141	/// the suffix for generated product names. It can be set via:
142	/// - Command-line: `--name <name>`
143	/// - Environment: `MOUNTAIN_ORIGINAL_BASE_NAME`
144	/// - Default: "Mountain"
145	#[clap(long, env = NameEnv, default_value = NameDefault)]
146	pub Name: String,
147
148	/// The prefix for the application's bundle identifier.
149	///
150	/// This field specifies the reverse domain prefix for the bundle
151	/// identifier. It can be set via:
152	/// - Command-line: `--prefix <prefix>`
153	/// - Environment: `MOUNTAIN_BUNDLE_ID_PREFIX`
154	/// - Default: "land.editor.binary"
155	#[clap(long, env = PrefixEnv, default_value = PrefixDefault)]
156	pub Prefix: String,
157
158	/// Flag or value indicating browser-specific build aspects.
159	///
160	/// When set to "true", enables browser-specific configuration and affects
161	/// the generated product name and bundle identifier.
162	#[clap(long, env = BrowserEnv)]
163	pub Browser: Option<String>,
164
165	/// Flag or value indicating bundling-specific aspects.
166	///
167	/// When set to "true", enables bundling-specific configuration and affects
168	/// the generated product name and bundle identifier.
169	#[clap(long, env = BundleEnv)]
170	pub Bundle: Option<String>,
171
172	/// Flag or value indicating compile-specific aspects.
173	///
174	/// When set to "true", enables compile-specific configuration and affects
175	/// the generated product name and bundle identifier.
176	#[clap(long, env = CompileEnv)]
177	pub Compile: Option<String>,
178
179	/// Flag or value indicating cleaning-specific aspects.
180	///
181	/// When set to "true", enables clean-specific configuration and affects
182	/// the generated product name and bundle identifier.
183	#[clap(long, env = CleanEnv)]
184	pub Clean: Option<String>,
185
186	/// Flag or value indicating debug-specific aspects.
187	///
188	/// When set to "true", enables debug-specific configuration and affects
189	/// the generated product name and bundle identifier. Also automatically
190	/// detected if the command contains "--debug".
191	#[clap(long, env = DebugEnv)]
192	pub Debug: Option<String>,
193
194	/// Information about a dependency, often 'org/repo' or a boolean string.
195	///
196	/// This field specifies dependency information that affects the generated
197	/// product name and bundle identifier. Can be:
198	/// - "true" for generic dependencies
199	/// - "org/repo" for specific repository dependencies
200	/// - Any custom string
201	#[clap(long, env = DependencyEnv)]
202	pub Dependency: Option<String>,
203
204	/// The Node.js environment (e.g., "development", "production").
205	///
206	/// This field specifies the Node.js runtime environment and affects the
207	/// generated product name and bundle identifier.
208	#[clap(long, env = NodeEnv)]
209	pub Environment: Option<String>,
210
211	/// Specifies the Node.js sidecar version to bundle (e.g., "22").
212	///
213	/// This field specifies which Node.js version should be bundled as a
214	/// sidecar binary with the application. The executable is sourced from
215	/// the Element/SideCar directory.
216	#[clap(long, env = NodeVersionEnv)]
217	pub NodeVersion: Option<String>,
218
219	/// The build command and its arguments to execute.
220	///
221	/// This field accepts all remaining command-line arguments as the build
222	/// command to execute after configuring the project files. This field
223	/// is required and must be provided as the final arguments.
224	///
225	/// Example: `pnpm tauri build`
226	#[clap(required = true, last = true)]
227	pub Command: Vec<String>,
228}
