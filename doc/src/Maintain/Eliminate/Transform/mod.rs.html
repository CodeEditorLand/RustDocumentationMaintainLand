4// Module: Transform - AST transformation pipeline
5//
6// Two entry points:
7//
8//   Run(source, options)
9//     Original behaviour: parse with syn, run the VisitMut eliminator,
10//     then attempt span-based text patching to preserve comments and
11//     whitespace. Falls back to prettyplease::unparse when span data is
12//     unavailable. Used by the Reformat path and by all existing unit tests
13//     in Inline.rs (which compare output against prettyplease-normalised
14//     expected values).
15//
16//   RunPreserve(source, options)
17//     Preserve-layout behaviour (default when Options.Reformat == false):
18//     identifies inlinable bindings via the same Collect/Safe/Count pipeline,
19//     then applies targeted text substitutions to the original source string
20//     without touching anything outside the affected lines.  Comments, blank
21//     lines, section banners, and the original indentation style survive
22//     unchanged.  Uses NO proc_macro2 span APIs so no extra Cargo features
23//     are required.
24//
25// Returns `Ok(None)` when no bindings were eliminated.
26//=============================================================================//

36// ---------------------------------------------------------------------------
37// Original entry point - span-based patch path with prettyplease fallback
38// (signature identical to Current; all Inline.rs tests call this function)
39// ---------------------------------------------------------------------------

41/// Parse `Source`, run up to [`super::Constant::MaxIterations`] elimination
42/// passes, then return the patched source text.
43///
44/// Preferred path: span-based text patching via `TryPatchSource` preserves
45/// inline comments and blank lines. Falls back to `prettyplease::unparse`
46/// when span-location data is unavailable.
47///
48/// Returns `Ok(None)` when no bindings were eliminated (caller can skip the
49/// write-back).
50pub fn Run(Source:&str, Options:&Definition::Options) -> Error::Result<Option<String>> {
51	let mut Ast:syn::File = syn::parse_str(Source).map_err(|E| Error::Error::Parse { Path:String::new(), Source:E })?;
52
53	let mut AnyChanged = false;
54
55	for _ in 0..super::Constant::MaxIterations {
56		let mut Eliminator = Inline::Eliminator::new(Options);
57
58		syn::visit_mut::visit_file_mut(&mut Eliminator, &mut Ast);
59
60		if Eliminator.Changed {
61			AnyChanged = true;
62		} else {
63			break;
64		}
65	}
66
67	if !AnyChanged {
68		return Ok(None);
69	}
70
71	// Attempt span-based text patching to preserve comments and whitespace.
72	// Build the patched output by comparing the original AST (re-parsed from
73	// Source so spans are relative to Source) against the mutated Ast.
74	//
75	// Strategy: re-parse Source into OriginalAst whose spans ARE anchored to
76	// Source bytes. Walk both ASTs in parallel, collecting (removed-let-span,
77	// substituted-ident-span, replacement-text) triples, then apply them.
78	if let Some(Patched) = TryPatchSource(Source, &Ast) {
79		return Ok(Some(Patched));
80	}
81
82	// Fallback: full reprint via prettyplease (drops comments / whitespace).
83	Ok(Some(prettyplease::unparse(&Ast)))
84}

90/// Attempt to reconstruct the output by splicing only the changed ranges.
91/// Returns None on any span resolution failure, triggering the fallback.
92fn TryPatchSource(Source:&str, MutatedAst:&syn::File) -> Option<String> {
93	// Re-parse Source so we have an AST whose Spans are anchored to Source.
94	let OriginalAst:syn::File = syn::parse_str(Source).ok()?;
95
96	let mut Patches:Vec<Patch::Patch> = Vec::new();
97
98	// Walk items in parallel. We only care about function bodies; other items
99	// (mod, use, struct, ...) are never touched by the eliminator.
100	for (OrigItem, MutItem) in OriginalAst.items.iter().zip(MutatedAst.items.iter()) {
101		if let (syn::Item::Fn(OrigFn), syn::Item::Fn(MutFn)) = (OrigItem, MutItem) {
102			CollectBlockPatches(&OrigFn.block, &MutFn.block, Source, &mut Patches)?;
103		}
104	}
105
106	if Patches.is_empty() {
107		// No spans resolved but AnyChanged was true - fall back.
108		return None;
109	}
110
111	// Sort by start offset and verify non-overlapping.
112	Patches.sort_by_key(|P| P.Start);
113
114	Patch::ApplyPatches(Source, &Patches)
115}

117/// Recursively compare OrigBlock (spans anchored to Source) and MutBlock
118/// (mutated AST, spans may be synthetic) and collect patches for any
119/// statements that differ.
120fn CollectBlockPatches(
121	OrigBlock:&syn::Block,
122	MutBlock:&syn::Block,
123	Source:&str,
124	Patches:&mut Vec<Patch::Patch>,
125) -> Option<()> {
126	// Walk MutBlock statements; for each one, find the corresponding
127	// statement(s) in OrigBlock by matching on identity (same kind, same
128	// initialiser text for lets). Statements the eliminator REMOVED will
129	// be present in OrigBlock but absent from MutBlock. Statements where
130	// an identifier was SUBSTITUTED will have a different token stream.
131
132	let mut OrigIdx = 0usize;
133
134	for MutStmt in &MutBlock.stmts {
135		// Advance OrigIdx until we find a statement that matches MutStmt
136		// or we exhaust OrigBlock.
137		while OrigIdx < OrigBlock.stmts.len() {
138			let OrigStmt = &OrigBlock.stmts[OrigIdx];
139			OrigIdx += 1;
140
141			if StmtTokensMatch(OrigStmt, MutStmt) {
142				// Same statement - no patch needed here.
143				break;
144			}
145
146			// OrigStmt is in the original but not in MutBlock at this
147			// position: it was a removed let. Emit a delete patch.
148			let (Start, End) = Patch::StmtLineRange(OrigStmt, Source)?;
149
150			Patches.push(Patch::Patch { Start, End, Replacement:String::new() });
151
152			// Now check if MutStmt corresponds to this OrigStmt after the
153			// removal - if not, continue consuming OrigBlock.
154			if StmtTokensMatch(&OrigBlock.stmts[OrigIdx - 1 + 1], MutStmt) {
155				break;
156			}
157		}
158
159		// Recurse into inner blocks.
160		CollectInnerBlockPatches(MutStmt, Source, Patches)?;
161	}
162
163	Some(())
164}

166/// Recurse into block-containing expression variants.
167fn CollectInnerBlockPatches(Stmt:&syn::Stmt, Source:&str, Patches:&mut Vec<Patch::Patch>) -> Option<()> {
168	use syn::{Expr, Stmt as S};
169
170	match Stmt {
171		S::Expr(Expr::Block(B), _) => {
172			for S_ in &B.block.stmts {
173				CollectInnerBlockPatches(S_, Source, Patches)?;
174			}
175		},
176
177		_ => {},
178	}
179
180	Some(())
181}

183/// Compare two statements by their token stream text.
184fn StmtTokensMatch(A:&syn::Stmt, B:&syn::Stmt) -> bool {
185	use quote::ToTokens;
186
187	let mut Ta = proc_macro2::TokenStream::new();
188	let mut Tb = proc_macro2::TokenStream::new();
189
190	A.to_tokens(&mut Ta);
191	B.to_tokens(&mut Tb);
192
193	Ta.to_string() == Tb.to_string()
194}

200/// Identify inlinable bindings via the same AST pipeline as `Run`, but apply
201/// the substitutions as targeted text edits so that every character outside
202/// the affected `let` binding and its single use-site is preserved verbatim.
203///
204/// Uses no `proc_macro2` span APIs; works on stable Rust with the dependency
205/// set already declared in `Cargo.toml`.
206///
207/// Returns `Ok(None)` when no bindings were eliminated.
208pub fn RunPreserve(Source:&str, Options:&Definition::Options) -> Error::Result<Option<String>> {
209	let mut Working = Source.to_owned();
210	let mut AnyChanged = false;
211
212	for _ in 0..super::Constant::MaxIterations {
213		match PreservePass(&Working, Options)? {
214			Some(Next) => {
215				Working = Next;
216				AnyChanged = true;
217			},
218
219			None => break,
220		}
221	}
222
223	if AnyChanged { Ok(Some(Working)) } else { Ok(None) }
224}
225
226/// One pass: parse `Working`, find the first inlinable binding, apply the
227/// text edit, return `Some(new_text)`. Returns `None` when nothing changed.
228fn PreservePass(Working:&str, Options:&Definition::Options) -> Error::Result<Option<String>> {
229	let Ast:syn::File = syn::parse_str(Working).map_err(|E| Error::Error::Parse { Path:String::new(), Source:E })?;
230
231	for Item in &Ast.items {
232		if let Some(Result) = TryItemPreserve(Item, Working, Options)? {
233			return Ok(Some(Result));
234		}
235	}
236
237	Ok(None)
238}
239
240fn TryItemPreserve(Item:&syn::Item, Working:&str, Options:&Definition::Options) -> Error::Result<Option<String>> {
241	match Item {
242		syn::Item::Fn(F) => TryBlockPreserve(&F.block, Working, Options),
243
244		syn::Item::Impl(I) => {
245			for ImplItem in &I.items {
246				if let syn::ImplItem::Fn(M) = ImplItem {
247					if let Some(R) = TryBlockPreserve(&M.block, Working, Options)? {
248						return Ok(Some(R));
249					}
250				}
251			}
252
253			Ok(None)
254		},
255
256		_ => Ok(None),
257	}
258}

260fn TryBlockPreserve(Block:&syn::Block, Working:&str, Options:&Definition::Options) -> Error::Result<Option<String>> {
261	let Candidates = Collect::Collect(Block, Options.InlineComments);
262
263	for Candidate in &Candidates {
264		if !Safe::IsSafe(&Candidate.Init, Options.MaxSize) {
265			continue;
266		}
267
268		let (RefCount, InClosure, InLoop) =
269			Count::CountReferences(&Candidate.Ident, &Block.stmts[Candidate.StmtIndex + 1..]);
270
271		if RefCount != 1 || InClosure || InLoop {
272			continue;
273		}
274
275		let StmtsAfter = &Block.stmts[Candidate.StmtIndex + 1..];
276		let SubstSiteOffset = Inline::FindSubstSite(StmtsAfter, &Candidate.Ident);
277		if !Safe::IsFreeVarSafe(&Candidate.Init, &StmtsAfter[..SubstSiteOffset]) {
278			continue;
279		}
280
281		// Clone downstream statements; substitute in-memory.
282		let mut UseStmts:Vec<syn::Stmt> = Block.stmts[Candidate.StmtIndex + 1..].to_vec();
283
284		if !Inline::SubstituteRef(&mut UseStmts, &Candidate.Ident, &Candidate.Init) {
285			continue;
286		}
287
288		// Render the ORIGINAL let-stmt and use-stmt to canonical text so we
289		// can locate them in `Working` by plain string search.
290		let LetText = StmtToText(&Block.stmts[Candidate.StmtIndex]);
291		let UseOrigText = StmtToText(&Block.stmts[Candidate.StmtIndex + 1 + SubstSiteOffset]);
292		let UseNewText = StmtToText(&UseStmts[SubstSiteOffset]);
293
294		// Locate the original let-stmt text in Working.
295		let Some(LetPos) = Working.find(&LetText) else {
296			continue;
297		};
298
299		// Locate the original use-stmt text - must appear AFTER the let.
300		let SearchFrom = LetPos + LetText.len();
301		let Some(UseOffset) = Working[SearchFrom..].find(&UseOrigText) else {
302			continue;
303		};
304
305		let UsePos = SearchFrom + UseOffset;
306
307		// Apply edits in reverse order (use comes later, so edit it first so
308		// the let-stmt byte positions remain valid).
309		let mut Out = Working.to_owned();
310
311		// 1. Replace the use-stmt with the substituted version.
312		Out.replace_range(UsePos..UsePos + UseOrigText.len(), &UseNewText);
313
314		// 2. Remove the let-stmt line (expand to include trailing newline).
315		let LetEnd = LetPos + LetText.len();
316		let ExpandedLetEnd = if LetEnd < Out.len() && Out.as_bytes()[LetEnd] == b'\n' {
317			LetEnd + 1
318		} else {
319			LetEnd
320		};
321
322		Out.replace_range(LetPos..ExpandedLetEnd, "");
323
324		return Ok(Some(Out));
325	}
326
327	// Recurse into directly nested blocks.
328	for Stmt in &Block.stmts {
329		if let Some(Nested) = StmtNestedBlock(Stmt) {
330			if let Some(R) = TryBlockPreserve(Nested, Working, Options)? {
331				return Ok(Some(R));
332			}
333		}
334	}
335
336	Ok(None)
337}

343/// Render a single `syn::Stmt` to its canonical text representation by
344/// wrapping it in a dummy function body and extracting the inner line(s).
345/// The wrapper indentation (one tab or 4 spaces from prettyplease) is
346/// stripped so the result is indentation-relative.
347fn StmtToText(Stmt:&syn::Stmt) -> String {
348	use quote::quote;
349
350	let Wrapped:syn::File = syn::parse_quote! { fn __d() { #Stmt } };
351	let Full = prettyplease::unparse(&Wrapped);
352
353	// Full looks like "fn __d() {\n    <stmt>\n}\n".
354	// Extract between first '{' and last '}'.
355	if let (Some(Open), Some(Close)) = (Full.find('{'), Full.rfind('}')) {
356		let Inner = Full[Open + 1..Close].trim_matches('\n');
357
358		return Inner
359			.lines()
360			.map(|L| L.strip_prefix('\t').or_else(|| L.strip_prefix("    ")).unwrap_or(L))
361			.collect::<Vec<_>>()
362			.join("\n");
363	}
364
365	Full
366}

368/// Extract a directly nested `Block` from a statement for recursion.
369fn StmtNestedBlock(Stmt:&syn::Stmt) -> Option<&syn::Block> {
370	if let syn::Stmt::Expr(Expr, _) = Stmt {
371		match Expr {
372			syn::Expr::Block(B) => return Some(&B.block),
373			syn::Expr::If(I) => return Some(&I.then_branch),
374			syn::Expr::Loop(L) => return Some(&L.body),
375			syn::Expr::While(W) => return Some(&W.body),
376			syn::Expr::ForLoop(F) => return Some(&F.body),
377			syn::Expr::Unsafe(U) => return Some(&U.block),
378			_ => {},
379		}
380	}
381	None
